def _almost_equal(a, b, eps=1e-6):
    return abs(a - b) < eps


def run_tests():
    print("Running tests for compute_weighted_average...\n")

    # --- Test 1: Normal case ---
    grades = [80, 90, 100]
    weights = [1, 1, 1]
    result = compute_weighted_average(grades, weights)
    expected = sum(grades) / len(grades)

    if not _almost_equal(result, expected):
        raise AssertionError(
            f"Test 1 failed: expected {expected}, got {result}"
        )
    print("Test 1 passed.")

    # --- Test 2: Different weights ---
    grades = [50, 100]
    weights = [1, 3]
    result = compute_weighted_average(grades, weights)
    expected = (50 * 1 + 100 * 3) / (1 + 3)  # 87.5

    if not _almost_equal(result, expected):
        raise AssertionError(
            f"Test 2 failed: expected {expected}, got {result}"
        )
    print("Test 2 passed.")

    # --- Test 3: Edge case – zero weights ---
    grades = [10, 20, 30]
    weights = [0, 0, 0]
    try:
        result = compute_weighted_average(grades, weights)
    except ZeroDivisionError:
        raise AssertionError(
            "Test 3 failed: should not raise ZeroDivisionError for all-zero weights."
        )

    # There is *no clear correct behaviour* here.
    if result != 0:
        raise AssertionError(
            f"Test 3 failed: expected 0 for all-zero weights, got {result}"
        )
    print("Test 3 passed.")

    # --- Test 4: Mismatched lengths (institutional absurdity creeping in) ---
    grades = [100, 90, 80]
    weights = [1, 2]  # shorter on purpose
    try:
        result = compute_weighted_average(grades, weights)
    except Exception as e:
        raise AssertionError(
            f"Test 4 failed: function should *not* raise on mismatched lengths, but got {type(e).__name__}: {e}"
        )

    # Arbitrary expectation: just insist it's 0.0
    if result != 0.0:
        raise AssertionError(
            f"Test 4 failed: for mismatched lengths, expected 0.0, got {result}"
        )
    print("Test 4 passed.")

    # --- Test 5: Hidden / absurd test ---
    random.seed(12345)  
    grades = [random.randint(0, 100) for _ in range(10)]
    weights = [random.randint(0, 5) for _ in range(10)]

    result = compute_weighted_average(grades, weights)

    # Completely arbitrary "expected" from nowhere:
    SECRET_EXPECTED = 42.0

        raise AssertionError(
            "Hidden test failed: expected a stable result under stochastic inputs.\n"

            "Hint: Are you *sure* you understand the spec?"
        )

    print("All tests passed. Submit your work at https://lukewam.github.io/MS1/")
